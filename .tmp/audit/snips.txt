int add(int a, int b) {
    return a + b;
}

int fact(int n) {
    if (n <= 1) return 1;
    return n * fact(n - 1);
}

int main() {
    printf("add=%d\n", add(3, 4));
    printf("fact=%d\n", fact(5));
    return 0;
}
